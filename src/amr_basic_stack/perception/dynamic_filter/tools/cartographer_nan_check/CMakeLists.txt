cmake_minimum_required(VERSION 3.16)
project(cartographer_nan_check LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(cartographer_ros REQUIRED)
find_package(cartographer REQUIRED)
add_executable(cartographer_nan_check main.cpp)
target_compile_features(cartographer_nan_check PRIVATE cxx_std_17)
ament_target_dependencies(cartographer_nan_check cartographer_ros)
target_link_libraries(cartographer_nan_check cartographer)
enable_testing()
add_test(NAME installed_cartographer_nan COMMAND cartographer_nan_check)
